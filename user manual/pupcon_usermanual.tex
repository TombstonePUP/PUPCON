\section{Introduction}
% Short description of the system and purpose of manual
PUCON (PUPCON Accreditation System) is a web-based platform designed to
streamline the accreditation process for coordinators, area chairs, and
evaluators. This manual explains how to access and use key features.

\section{System Requirements}
\begin{itemize}
    \item \textbf{Supported Browsers:} Google Chrome, Mozilla Firefox, Microsoft Edge
    \item \textbf{Internet:} Stable connection recommended (\textgreater= 5 Mbps)
    \item \textbf{Device:} Desktop or Laptop (recommended resolution: 1366x768 or higher)
\end{itemize}

\section{Accessing the System}
\begin{enumerate}
    \item Open your web browser.
    \item Navigate to: \url{https://pucon.example.com}.
    \item Enter your registered \textbf{Username} and \textbf{Password}.
    \item Click \textbf{Login}.
\end{enumerate}

\subsection{Forgot Password}
If you forgot your password, click the \textbf{Forgot Password} link and follow the
recovery instructions sent to your registered email.

\section{User Roles}
\begin{description}[leftmargin=!,labelwidth=3cm]
    \item[Accreditation Coordinator] Oversees and manages the entire accreditation process.
    \item[Area Chair] Uploads, reviews, and monitors documents for specific areas.
    \item[Evaluator] Reviews submissions and provides feedback.
    \item[Admin] Manages users and system settings.
\end{description}

\section{Main Features}
\subsection{Dashboard}
Displays pending tasks, recent activities, and system notifications.

\subsection{Document Management}
Upload, view, and download accreditation-related files. Organize documents by accreditation area.

\subsection{Evaluation Module}
Submit evaluation forms and view consolidated feedback and results.

\subsection{Reports}
Generate and download summary reports. Filter by date, area, or evaluator.

\section{Common Tasks}
\subsection{Uploading a Document}
\begin{enumerate}
    \item Go to \textbf{Document Management}.
    \item Select the appropriate \textbf{Accreditation Area}.
    \item Click \textbf{Upload File}.
    \item Browse and select your file.
    \item Click \textbf{Submit}.
\end{enumerate}

\subsection{Viewing Evaluation Results}
\begin{enumerate}
    \item Navigate to \textbf{Evaluation Module}.
    \item Select your submitted documents.
    \item Click \textbf{View Results} to see feedback and scores.
\end{enumerate}

\section{Troubleshooting}
\begin{itemize}
    \item \textbf{Login Issues:} Check your username/password. Use \textbf{Forgot Password} if needed.
    \item \textbf{File Upload Fails:} Ensure the file is within allowed size and format.
    \item \textbf{Page Not Loading:} Refresh the page or check your internet connection.
\end{itemize}

\section{Support}
For assistance, contact the PUCON Support Team at:
\begin{itemize}
    \item Email: \href{mailto:support@pucon.example.com}{support@pucon.example.com}
    \item Phone: +63-XXX-XXX-XXXX
\end{itemize}
